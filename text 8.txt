local function safeRequire(getInst)
    local ok, result = pcall(function()
        local inst = getInst()
        if not inst then error("nil instance") end
        return require(inst)
    end)
    if not ok then warn("[safeRequire] failed: " .. tostring(result)) end
    return ok and result or nil
end

    Original    = nil,
    Clone       = nil,

local function getWalkSpeed()
    local char     = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    local speed    = humanoid and humanoid.WalkSpeed or 16

    if SpeedHack.Original and SpeedHack.Original.Health > 0 then
        speed = math.min(speed, SpeedHack.Original.WalkSpeed)
    end

    local ok, result = pcall(function()
        local stats    = LocalPlayer:FindFirstChild("leaderstats")
        local speedStat= stats and stats:FindFirstChild("Speed")
        local util     = require(ReplicatedStorage.Shared.Util.TreadmillUtil)
        return speedStat and util.SpeedPowerToWalkSpeed(speedStat.Value) or nil
    end)
    if ok and type(result) == "number" and result > 0 then
        speed = math.min(speed, result)
    end

    return speed
end

local function setControls(humanoid)
    pcall(function()
        local scripts = LocalPlayer:FindFirstChild("PlayerScripts")
        local module  = scripts and scripts:FindFirstChild("PlayerModule")
        if module then
            local controls = require(module):GetControls()
            if type(controls) == "table" then controls.humanoid = humanoid end
        end
    end)
end

local function refreshAnimator(char)
    local animate = char and char:FindFirstChild("Animate")
    if animate and animate:IsA("LocalScript") then
        task.spawn(function()
            animate.Enabled = false
            task.wait()
            animate.Enabled = true
        end)
    end
end

local function disconnectSpeedHack()
    for _, conn in ipairs(SpeedHack.Connections) do
        pcall(function() conn:Disconnect() end)
    end
    table.clear(SpeedHack.Connections)
end

local isGroundedStates = {
    [Enum.HumanoidStateType.Running]          = true,
    [Enum.HumanoidStateType.RunningNoPhysics] = true,
    [Enum.HumanoidStateType.Landed]           = true,
}

local function isGrounded(humanoid)
    if not humanoid or humanoid.Health <= 0 then return false end
    if humanoid.FloorMaterial == Enum.Material.Air then return false end
    return isGroundedStates[humanoid:GetState()] == true
end

local swapPending = false
local function swapHumanoid()
    local char     = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health <= 0 then return end
    if SpeedHack.Clone and SpeedHack.Clone.Parent == char then return end

    if not isGrounded(humanoid) then
        if not swapPending then
            swapPending = true
            task.delay(0.3, function()
                swapPending = false
                swapHumanoid()
            end)
        end
        return
    end

    local clone        = humanoid:Clone()
    humanoid.Parent    = nil
    clone.Parent       = char
    workspace.CurrentCamera.CameraSubject = clone
    setControls(clone)
    refreshAnimator(char)
    SpeedHack.Original = humanoid
    SpeedHack.Clone    = clone

    table.insert(SpeedHack.Connections,
        humanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
            if clone.Parent ~= nil then
                clone.WalkSpeed = humanoid.WalkSpeed
            end
        end)
    )

    local animatorOrig  = humanoid:FindFirstChildOfClass("Animator")
    local animatorClone = clone:FindFirstChildOfClass("Animator")
    if animatorOrig and animatorClone then
        table.insert(SpeedHack.Connections,
            animatorOrig.AnimationPlayed:Connect(function(track)
                local anim = track.Animation
                if not anim or clone.Parent == nil then return end
                local ok, newTrack = pcall(function() return animatorClone:LoadAnimation(anim) end)
                if not ok or not newTrack then return end
                pcall(function()
                    newTrack.Priority = track.Priority
                    newTrack.Looped   = track.Looped
                    newTrack:Play(0.05, math.max(track.WeightTarget, 0.01), track.Speed)
                end)
                local stopConn
                stopConn = track.Stopped:Connect(function()
                    stopConn:Disconnect()
                    pcall(function() newTrack:Stop(0.1) end)
                end)
            end)
        )
    end

    table.insert(SpeedHack.Connections,
        clone.Died:Connect(function()
            disconnectSpeedHack()
            SpeedHack.Original = nil
            SpeedHack.Clone    = nil
            local currentChar = LocalPlayer.Character
            if currentChar and humanoid.Parent == nil then
                humanoid.Parent = currentChar
                workspace.CurrentCamera.CameraSubject = humanoid
                setControls(humanoid)
            end
            pcall(function() clone:Destroy() end)
            humanoid.Health = 0
        end)
    )
end

local function restoreHumanoid()
    disconnectSpeedHack()
    local char     = LocalPlayer.Character
    local original = SpeedHack.Original
    local clone    = SpeedHack.Clone
    SpeedHack.Original = nil
    SpeedHack.Clone    = nil
    if original and clone and char and original.Parent == nil and clone.Parent == char then
        original.Parent = char
        workspace.CurrentCamera.CameraSubject = original
        setControls(original)
        pcall(function() clone:Destroy() end)
        refreshAnimator(char)
    end
end

local function startSpeedHack()
    swapHumanoid()
    local elapsed = 0
    SpeedHack.Heartbeat = RunService.Heartbeat:Connect(function(dt)
        elapsed += dt
        local char    = LocalPlayer.Character
        local missing = not (SpeedHack.Clone and char and SpeedHack.Clone.Parent == char)
        if (missing and 0.25 or 3) <= elapsed then
            elapsed = 0
            swapHumanoid()
        end
    end)
    SpeedHack.CharAdded = LocalPlayer.CharacterAdded:Connect(function(char)
        disconnectSpeedHack()
        SpeedHack.Original = nil
        SpeedHack.Clone    = nil
        task.spawn(function()
            char:WaitForChild("Humanoid", 10)
            task.wait(1)
            if SpeedHack.Heartbeat and LocalPlayer.Character == char then
                swapHumanoid()
            end
        end)
    end)
end

local function stopSpeedHack()
    if SpeedHack.Heartbeat then SpeedHack.Heartbeat:Disconnect(); SpeedHack.Heartbeat = nil end
    if SpeedHack.CharAdded  then SpeedHack.CharAdded:Disconnect();  SpeedHack.CharAdded  = nil end
    restoreHumanoid()
end

local function getNearestPrompt(pos, radius)
    local bestPrompt, bestDist = nil, radius
    for _, child in ipairs(workspace:GetChildren()) do
        if child.Name == "SmartPromptPart" and child:IsA("BasePart") then
            local prompt = child:FindFirstChild("CarryAreaEgg")
            if prompt and prompt:IsA("ProximityPrompt") then
                local dist = (child.Position - pos).Magnitude
                if dist < bestDist then
                    bestPrompt = prompt
                    bestDist   = dist
                end
            end
        end
    end
    return bestPrompt
end

local function isEggHeldByMe(uid)
    local char = LocalPlayer.Character
    if type(uid) ~= "string" or not char then return false end
    local eggModel = workspace:FindFirstChild(uid)
    if not eggModel then return false end
    for _, desc in ipairs(eggModel:GetDescendants()) do
        if desc:IsA("WeldConstraint") or desc:IsA("JointInstance") then
            local ok, p0, p1 = pcall(function() return desc.Part0, desc.Part1 end)
            if ok then
                if (p0 and p0:IsDescendantOf(char)) or (p1 and p1:IsDescendantOf(char)) then
                    return true
                end
            end
        end
    end
    return false
end

local function getSnapshot()
    local list   = {}
    local ok, rf = pcall(function() return findRemote("RF/EggWorld/AskFieldEggSnapshot") end)
    if not ok or not rf then return list end
    local ok2, result = pcall(function() return rf:InvokeServer() end)
    if not ok2 or type(result) ~= "table" then return list end
    local records = result.Records
    if type(records) ~= "table" then return list end
    for _, rec in pairs(records) do
        if type(rec) == "table"
        and type(rec.Uid) == "string"
        and (rec.State == "Slot" or rec.State == "Dropped")
        and typeof(rec.BottomCFrame) == "CFrame"
        then
            list[#list + 1] = rec
        end
    end
    return list
end

local function getEggPosition(uid)
    for _, rec in ipairs(getSnapshot()) do
        if rec.Uid == uid then return rec.BottomCFrame.Position end
    end
    return nil
end

local function describeEgg(rec)
    local directory = Assets and type(Assets) == "table" and Assets.Directory or nil
    local entry     = type(directory) == "table" and directory[tostring(rec.AssetCategory)] or nil
    local info      = {
        Uid  = rec.Uid,
        Cat  = tostring(rec.AssetCategory),
        Name = tostring(rec.AssetCategory),
        Icon = "",
        Rarity = 0,
        Value  = 0,
    }
    if type(entry) == "table" then
        info.Name = tostring(entry.DisplayName or rec.AssetCategory)
        local icon = entry.Icon
        if icon ~= nil and tostring(icon) ~= "" then
            icon      = tostring(icon)
            info.Icon = tonumber(icon) and ("rbxassetid://" .. icon) or icon
        end
        if type(entry.Rarity) == "table" then
            info.Rarity = tonumber(entry.Rarity.RarityNumber or entry.Rarity.Rank) or 0
        end
        local scale      = tonumber(rec.AssetScale) or 1
        local scaleFactor= scale > 5 and (scale / 5) ^ 1.2 * 19.637875755794113 or scale ^ 1.85
        local mutMult    = 1
        if Mutations and type(Mutations) == "table" and type(Mutations.EarningsFor) == "function" then
            local ok, val = pcall(Mutations.EarningsFor, type(rec.Mutations) == "table" and rec.Mutations or {})
            if ok and type(val) == "number" then mutMult = val end
        end
        info.Value = (tonumber(entry.EarningRate) or 0) * scaleFactor * mutMult
    end
    return info
end

local function formatMoney(value)
    local units = { {1e12,"N"}, {1e9,"T"}, {1e6,"Tr"}, {1e3,"K"} }
    for _, u in ipairs(units) do
        if value >= u[1] then
            return string.format("$%.2f%s/s", value / u[1], u[2])
        end
    end
    return string.format("$%d/s", math.floor(value + 0.5))
end

local fpsGeneration = 0

local function enableFpsCap()
    if typeof(setfpscap) ~= "function" then return end
    fpsGeneration += 1
    local gen = fpsGeneration
    task.spawn(function()
        local flip  = true
        local start = os.clock()
        while fpsGeneration == gen and os.clock() - start < 60 do
            pcall(setfpscap, flip and FpsCycle[1] or FpsCycle[2])
            flip = not flip
            task.wait(FpsCycle[3])
        end
    end)
end

local function disableFpsCap()
    fpsGeneration += 1
    if typeof(setfpscap) == "function" then pcall(setfpscap, 240) end
end

    if not char then return end

    char.Archivable    = true

        for _, desc in ipairs(clone:GetDescendants()) do
            if desc:IsA("LuaSourceContainer") or desc:IsA("Humanoid") then

                desc.CanCollide  = false
                desc.CanTouch    = false
                desc.CanQuery    = false
            end
        end
        clone.Name   = "Clone"
        clone.Parent = workspace
        activeClone  = clone
    end
end

-- ============================================================

local dangerCache    = {}
local dangerCacheTime= 0

local function getDangerZones()
    if os.clock() - dangerCacheTime < 1 then return dangerCache end
    dangerCacheTime = os.clock()
    local zones     = {}

    local function addZone(inst)
        local ok, cf, sz = pcall(function()
            if inst:IsA("Model")    then return inst:GetBoundingBox()
            elseif inst:IsA("BasePart") then return inst.CFrame, inst.Size end
        end)
        if not ok or not cf or not sz then return end
        local half  = Vector3.new(math.abs(sz.X), 0, math.abs(sz.Z)) * 0.5
        local rot   = (cf - cf.Position):VectorToWorldSpace(half)
        local rx    = math.max(math.abs(rot.X), half.X, half.Z)
        local rz    = math.max(math.abs(rot.Z), half.X, half.Z)
        zones[#zones + 1] = {
            MinX = cf.Position.X - rx, MaxX = cf.Position.X + rx,
            MinZ = cf.Position.Z - rz, MaxZ = cf.Position.Z + rz,
        }
    end

    local function isDangerous(name)
        if name == "ScrambleLocalVisuals" or name == "DrScrambleEvent" then return false end
        name = string.lower(name)
        return string.find(name, "portal",    1, true)
            or string.find(name, "teleport",  1, true)
            or string.find(name, "mech",      1, true)
            or string.find(name, "arena",     1, true)
            or string.find(name, "scramble",  1, true)
    end

    for _, child in ipairs(workspace:GetChildren()) do
        if (child:IsA("Model") or child:IsA("BasePart") or child:IsA("Folder")) and isDangerous(child.Name) then
            if child:IsA("Folder") then
                for _, inst in ipairs(child:GetChildren()) do addZone(inst) end
            else
                addZone(child)
            end
        end
    end

    local buildFolder = workspace:FindFirstChild("World")
    buildFolder = buildFolder and buildFolder:FindFirstChild("Build")
    if buildFolder then
        for _, child in ipairs(buildFolder:GetChildren()) do
            if isDangerous(child.Name) then
                for _, inst in ipairs(child:GetChildren()) do addZone(inst) end
            end
        end
    end

    dangerCache = zones
    return zones
end

local function avoidDangers(from, to)
    for _, zone in ipairs(getDangerZones()) do
        local x0 = zone.MinX - 12
        local x1 = zone.MaxX + 12
        local z0 = zone.MinZ - 12
        local z1 = zone.MaxZ + 12

        local inside = from.X >= x0 and from.X <= x1 and from.Z >= z0 and from.Z <= z1
        if not inside then
            local tMin, tMax, hit = 0, 1, true
            for _, axis in ipairs({
                { from.X, to.X - from.X, x0, x1 },
                { from.Z, to.Z - from.Z, z0, z1 },
            }) do
                local p, d, lo, hi = axis[1], axis[2], axis[3], axis[4]
                if math.abs(d) < 1e-6 then
                    if p < lo or p > hi then hit = false end
                else
                    local ta, tb = (lo - p) / d, (hi - p) / d
                    if ta > tb then ta, tb = tb, ta end
                    tMin = math.max(tMin, ta)
                    tMax = math.min(tMax, tb)
                    if tMin > tMax then hit = false end
                end
            end
            if hit then
                local zLow   = z0 - 2
                local zHigh  = z1 + 2
                local zPick  = math.abs(from.Z - zLow) <= math.abs(from.Z - zHigh) and zLow or zHigh
                if zPick < -440 or zPick > -290 then
                    zPick = (zPick == zLow) and zHigh or zLow
                end
                local xPick  = math.abs(from.X - x0) <= math.abs(from.X - x1) and x0 or x1
                if math.abs(from.Z - zPick) < 3 then
                    xPick = math.abs(to.X - x0) <= math.abs(to.X - x1) and x0 or x1
                end
                return Vector3.new(xPick, to.Y, zPick)
            end
        end
    end
    return to
end

local function runToEgg(uid, eggPos)
    local lineX      = getSeparationLine()
    local homePos    = getHomePosition()
    local char       = LocalPlayer.Character
    local humanoid   = char and char:FindFirstChildOfClass("Humanoid")

    if humanoid then
        humanoid.PlatformStand = false
        if char:FindFirstChildWhichIsA("Tool") then
            pcall(function() humanoid:UnequipTools() end)
        end
    end

    local root = getRootPart()
    if not root then return false end

    local stage = "field"
    if root.Position.X < lineX - 2 then
        local distHome = Vector3.new(root.Position.X - homePos.X, 0, root.Position.Z - homePos.Z).Magnitude
        if distHome > 20 then stage = "safe" end
    end

    local startTime    = os.clock()
    local stuckCheckAt = os.clock()
    local lastPos      = root.Position
    local lastTakeAt   = 0

    while os.clock() - startTime < 120 and not CarryState.Cancelled do
        root = getRootPart()
        if not root then return false end

        local toEgg = Vector3.new(eggPos.X - root.Position.X, 0, eggPos.Z - root.Position.Z)
        if stage == "field" and toEgg.Magnitude <= 2.5 then break end

        local target = eggPos
        if stage == "safe" then
            local distHome = Vector3.new(homePos.X - root.Position.X, 0, homePos.Z - root.Position.Z).Magnitude
            if distHome <= 6 then
                stage = "field"
            else
                target = homePos
            end
        end

        local waypoint = avoidDangers(root.Position, target)
        local flat     = Vector3.new(waypoint.X - root.Position.X, 0, waypoint.Z - root.Position.Z)
        local dir      = flat.Magnitude > 0.01 and flat.Unit or Vector3.zero
        local speed    = math.max(getWalkSpeed() * Config.SpeedCap, 8)
        local velocity = dir * math.min(speed, flat.Magnitude / 0.05)

        pcall(function()
            root.AssemblyLinearVelocity = Vector3.new(velocity.X, root.AssemblyLinearVelocity.Y, velocity.Z)
            if humanoid and dir.Magnitude > 0 then humanoid:Move(dir, false) end
        end)

        if os.clock() - stuckCheckAt >= 1.5 then
            if (root.Position - lastPos).Magnitude < 3 and humanoid and toEgg.Magnitude > 15 then
                pcall(function() humanoid.Jump = true end)
            end
            lastPos      = root.Position
            stuckCheckAt = os.clock()
        end

        if stage == "field" and toEgg.Magnitude <= 9 and os.clock() - lastTakeAt > 0.1 then
            lastTakeAt = os.clock()
            takeEgg(uid)
        end

        RunService.Heartbeat:Wait()
    end

    root = getRootPart()
    if root then
        pcall(function()
            root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
            if humanoid then humanoid:Move(Vector3.zero, false) end
        end)
    end

    local finalDist = root and Vector3.new(eggPos.X - root.Position.X, 0, eggPos.Z - root.Position.Z).Magnitude or math.huge
    return CarryState.IsCarrying or finalDist <= 6
end

local function getLiveEggPosition(uid, cache)
    local node = workspace:FindFirstChild(uid)
    local slots= workspace:FindFirstChild("AreaEggSlotsClient")
    node = node or (slots and slots:FindFirstChild(uid))
    if node then
        local ok, pos = pcall(function() return node:GetPivot().Position end)
        if ok and pos then
            cache.Pos = pos
            cache.At  = os.clock()
            return pos
        end
    end
    if os.clock() - cache.At >= 0.5 then
        cache.At  = os.clock()
        cache.Pos = getEggPosition(uid) or cache.Pos
    end
    return cache.Pos
end

local function grabEgg(uid, timeout)
    local cache    = { At = 0 }
    local elapsed  = 0
    local sinceLastFire = 1

    while not CarryState.IsCarrying and elapsed < timeout and not CarryState.Cancelled do
        local root    = getRootPart()
        local eggPos  = getLiveEggPosition(uid, cache)
        local dt      = math.max(RunService.Heartbeat:Wait(), 1 / 240)
        elapsed          += dt
        sinceLastFire    += dt

        if root and eggPos then
            local delta = eggPos - root.Position
            if delta.Magnitude > 2 then
                local maxSpeed = math.max(getWalkSpeed() * Config.SpeedCap, 16)
                local velocity = delta / math.max(0.08, dt)
                if velocity.Magnitude > maxSpeed then velocity = velocity.Unit * maxSpeed end
                pcall(function()
                    root.AssemblyLinearVelocity  = velocity + Vector3.new(0, workspace.Gravity * dt * 0.5, 0)
                    root.AssemblyAngularVelocity = Vector3.zero
                end)
            end
            if sinceLastFire >= Config.GrabInterval then
                sinceLastFire = 0
                local prompt  = getNearestPrompt(eggPos - Vector3.new(0, 3, 0), 10)
                if prompt and typeof(fireproximityprompt) == "function" then
                    pcall(function() prompt.HoldDuration = 0 end)
                    pcall(fireproximityprompt, prompt)
                end
                task.spawn(takeEgg, uid)
            end
        elseif sinceLastFire >= Config.GrabInterval then
            sinceLastFire = 0
            task.spawn(takeEgg, uid)
        end
    end

    local root = getRootPart()
    if root then pcall(function() root.AssemblyLinearVelocity = Vector3.zero end) end
    return CarryState.IsCarrying and CarryState.EggUid == uid
end

local function regrabEgg(uid)
    if CarryState.IsCarrying then return true end
    local root    = getRootPart()
    local eggPos  = getEggPosition(uid)
    if root and eggPos then
        local dist = Vector3.new(root.Position.X - eggPos.X, 0, root.Position.Z - eggPos.Z).Magnitude
        if dist > Config.RegrabFar then
            setStatus(Text.EggFar)
            teleportTo(eggPos + Vector3.new(0, 3, 0))
        else
            setStatus(Text.EggNear)
        end
    end
    return grabEgg(uid, 3)
end

-- ============================================================
-- DROP AND REGRAB  (shared between steal + bay)
-- ============================================================

local function dropAndRegrab(uid, isCancelled)
    isCancelled = isCancelled or function() return CarryState.Cancelled end

    setStatus(Text.DropEgg)
    local preDropPos = getEggPosition(uid)

    -- drop
    local delay = 0
    while delay < Config.DropDelay do delay += RunService.Heartbeat:Wait() end
    dropEgg()

    -- poll for egg to appear in workspace (max 0.25s)
    local dropPos  = nil
    local pollTime = 0
    while pollTime < 0.25 and not isCancelled() do
        pollTime += RunService.Heartbeat:Wait()
        local node  = workspace:FindFirstChild(uid)
        local slots = workspace:FindFirstChild("AreaEggSlotsClient")
        node = node or (slots and slots:FindFirstChild(uid))
        if node then
            local ok, pos = pcall(function() return node:GetPivot().Position end)
            if ok and pos then dropPos = pos; break end
        end
    end
    dropPos = dropPos or preDropPos
    if isCancelled() then return false end

    -- TP onto egg + zero velocity
    if dropPos then teleportTo(dropPos + Vector3.new(0, 2, 0)) end
    local root = getRootPart()
    if root then
        pcall(function()
            root.AssemblyLinearVelocity  = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end

    setStatus(Text.RegrabEgg)

    -- spam grab for 0.4s max, exit immediately on carry
    local spamTime = 0
    while not CarryState.IsCarrying and spamTime < 0.4 and not isCancelled() do
        spamTime += RunService.Heartbeat:Wait()
        task.spawn(takeEgg, uid)
        local epos = getEggPosition(uid)
        if epos then
            local prompt = getNearestPrompt(epos - Vector3.new(0, 3, 0), 10)
            if prompt and typeof(fireproximityprompt) == "function" then
                pcall(function() prompt.HoldDuration = 0 end)
                pcall(fireproximityprompt, prompt)
            end
        end
    end

    -- fallback
    if not (CarryState.IsCarrying and CarryState.EggUid == uid) then
        if not regrabEgg(uid) then
            setStatus(Text.CantRegrab)
            return false
        end
    end

    return CarryState.IsCarrying and CarryState.EggUid == uid
end

local function getGroundY(pos, fallback)
    local y = fallback
    pcall(function()
        local params = RaycastParams.new()
        params.FilterType                  = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances  = { LocalPlayer.Character }
        params.IgnoreWater                 = true
        local hit = workspace:Raycast(pos + Vector3.new(0, 60, 0), Vector3.new(0, -140, 0), params)
        if hit and hit.Material ~= Enum.Material.Water and math.abs(hit.Position.Y - fallback) < 40 then
            y = hit.Position.Y + 3.5
        end
    end)
    return y
end

-- ============================================================
-- CARRY SPEED
-- ============================================================

local function getCarrySpeed(dist)
    local base       = getWalkSpeed()
    local maxCarry   = base * math.min(Config.CarryRatio, Config.SpeedCap) * CarryState.SpeedMult
    local fastSpeed  = maxCarry * 1.5
    local extraDist  = 5.5 * maxCarry
    local limit      = fastSpeed
    if dist and dist > extraDist then
        limit = math.min(fastSpeed, maxCarry * dist / (dist - extraDist))
    end
    return math.min(
        math.max(math.min(maxCarry * Config.EasyRatio, limit), maxCarry),
        math.max(base * Config.SpeedCap, maxCarry)
    )
end

local function runHome(lineX, lineZ, isCancelled)
    isCancelled  = isCancelled or function() return CarryState.Cancelled end
    local t0     = os.clock()
    local home   = getHomePosition()
    local crossX = lineX - 7
    local height = Config.Height
    local share  = math.clamp(Config.ClimbShare, 0.1, 0.9)
    local slope  = height * math.sqrt(1 - share * share) / share
    local char   = LocalPlayer.Character
    local hu     = char and char:FindFirstChildOfClass("Humanoid")
    if hu then hu.PlatformStand = false end

    local root = getRootPart()
    if root and char and height > 0.5 and home.Y + height - 2 > root.Position.Y then
        pcall(function()
            char:PivotTo(CFrame.new(Vector3.new(root.Position.X, home.Y + height, root.Position.Z)) * root.CFrame.Rotation)
            root.AssemblyLinearVelocity  = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end)
    end

    local rootNow = getRootPart()
    local dist    = rootNow and Vector3.new(rootNow.Position.X - home.X, 0, rootNow.Position.Z - home.Z).Magnitude or 0
    local speed   = getCarrySpeed(dist + math.max(0, height) * 2)
    local last    = os.clock()
    local elapsed = 0
    setStatus(Text.GoHome)

    while CarryState.DeliverTime < t0 and not isCancelled() and elapsed < 45 do
        root = getRootPart()
        if not root then return false end
        local now = os.clock()
        local dt  = math.max(now - last, 1 / 240)
        last      = now
        elapsed  += dt

        local atHome   = root.Position.X <= crossX + 2
        local target   = atHome and home or Vector3.new(crossX, home.Y, lineZ)
        if atHome then
            local toHome = Vector3.new(home.X - root.Position.X, 0, home.Z - root.Position.Z).Magnitude
            if toHome < 2 then break end
        end

        local waypoint  = avoidDangers(root.Position, target)
        local flat      = Vector3.new(waypoint.X - root.Position.X, 0, waypoint.Z - root.Position.Z)
        local fromLine  = atHome and 0 or math.max(0, root.Position.X - crossX)
        local targetY   = home.Y + height
        if atHome or fromLine <= slope then
            targetY = home.Y + height * math.clamp(fromLine / math.max(slope, 1), 0, 1)
        end

        local vy = math.clamp((targetY - root.Position.Y) / 0.12, -speed * share, speed * share)
        local hz = math.sqrt(math.max(speed * speed - vy * vy, 0))
        local v  = flat.Magnitude > 0.01 and flat.Unit * math.min(hz, flat.Magnitude / 0.05) or Vector3.zero

        pcall(function() root.AssemblyLinearVelocity = Vector3.new(v.X, vy, v.Z) end)
        RunService.Heartbeat:Wait()
    end

    if hu then pcall(function() hu:Move(Vector3.zero, false) end) end

    local settle = 0
    while settle < 2 and CarryState.DeliverTime < t0 and CarryState.IsCarrying and not isCancelled() do
        settle += RunService.Heartbeat:Wait()
    end

    if CarryState.IsCarrying and CarryState.DeliverTime < t0 and not isCancelled() then
        teleportTo(Vector3.new(615.70, 70.57, -395.30))
        task.wait(0.2)
        dropEgg()
        local waited = 0
        while CarryState.DeliverTime < t0 and waited < 2 do
            waited += RunService.Heartbeat:Wait()
        end
    end

    local finalRoot = getRootPart()
    local nearHome  = finalRoot and Vector3.new(finalRoot.Position.X - home.X, 0, finalRoot.Position.Z - home.Z).Magnitude < 25
    return CarryState.DeliverTime >= t0 or (nearHome and not CarryState.IsCarrying)
end

-- ============================================================
-- INSTANT TELEPORT DELIVERY
-- ============================================================

local function deliverByTP(uid, isCancelled)
    isCancelled = isCancelled or function() return CarryState.Cancelled end
    local t0    = os.clock()
    local lineX, lineY = getSeparationLine()
    local root  = getRootPart()
    if not root then return false end

    local lineZ    = -395.30
    local landPos  = Vector3.new(lineX - Config.LandOffset, lineY + 3.35, lineZ)
    local hopSize  = math.max(getWalkSpeed() * Config.HopRatio, Config.HopMin)
    local hopY     = root.Position.Y + Config.HopLift
    local currentX = root.Position.X
    local retries  = 0
    CarryState.CleanupFn = nil

    makeClone()
    enableFpsCap()

    local skipHops = currentX <= lineX + 2
    if not skipHops then
        while currentX - hopSize > landPos.X and not isCancelled() do
            if not CarryState.IsCarrying then
                if not regrabEgg(uid) then break end
                local r = getRootPart()
                if r then currentX = math.min(currentX, r.Position.X) end
                continue
            end

            local nextX     = currentX - hopSize
            setStatus(Text.FlyingHome)
            local pullRef   = CarryState.PullTime
            local hopElapsed= 0

            while hopElapsed < Config.HopGap do
                teleportTo(Vector3.new(nextX, hopY, lineZ))
                hopElapsed += RunService.Heartbeat:Wait()
            end

            local checkRoot = getRootPart()
            local wasPulled = CarryState.PullTime > pullRef
                or (checkRoot ~= nil and (
                    checkRoot.Position.X - nextX > 10
                    or checkRoot.AssemblyLinearVelocity.Magnitude > 150
                ))

            if wasPulled and CarryState.IsCarrying and not isCancelled() then
                retries += 1
                Config.HopGap = math.min(Config.HopGapMax, Config.HopGap + 0.02)
                if retries > Config.HopRetries then break end
                setStatus(Text.PulledBack .. retries .. "/" .. Config.HopRetries)
                local hold = 0
                while hold < 0.12 and not isCancelled() do
                    teleportTo(Vector3.new(currentX, hopY, lineZ))
                    hold += RunService.Heartbeat:Wait()
                end
            else
                currentX    = nextX
                Config.HopGap = math.max(Config.HopGapMin, Config.HopGap - 0.01)
            end
        end
    end

    setStatus(Text.LandNear)
    landPos = Vector3.new(landPos.X, getGroundY(landPos, landPos.Y), landPos.Z)

    local function land()
        local r = getRootPart()
        if r and r.Position.X <= landPos.X + 1 and r.Position.Y > landPos.Y - 25 then
            pcall(function() r.AssemblyLinearVelocity = Vector3.zero end)
            return
        end
        teleportTo(landPos)
    end

    land()
    for attempt = 1, 2 do
        local settle = 0
        while settle < 0.04 and not isCancelled() do settle += RunService.Heartbeat:Wait() end
        local r = getRootPart()
        if CarryState.IsCarrying and r and (r.Position.X - landPos.X > 12 or r.Position.Y < landPos.Y - 25) then
            setStatus(Text.LandAgain .. attempt)
            land()
        else
            break
        end
    end

    destroyClone()

    if CarryState.IsCarrying and not isCancelled() then
        if not dropAndRegrab(uid, isCancelled) then
            disableFpsCap()
            if isCancelled() then return false end
            return false
        end
        destroyClone()
    end

    local delivered = false
    if CarryState.IsCarrying then
        delivered = runHome(lineX, lineZ, isCancelled)
    end

    disableFpsCap()
    if isCancelled() then return false end

    local noEgg = not CarryState.IsCarrying
    return delivered or CarryState.DeliverTime >= t0 or noEgg
end

-- ============================================================
-- AUTO STEAL
-- ============================================================

local function stealAndDeliver(uid, onFinished)
    if CarryState.Busy then return end
    CarryState.Busy      = true
    CarryState.Cancelled = false

    task.spawn(function()
        local ok, err = pcall(function()
            setStatus(Text.FindEgg)
            local eggPos = getEggPosition(uid)
            if not eggPos then setStatus(Text.EggGone); return end

            if CarryState.IsCarrying and CarryState.EggUid ~= uid then
                dropEgg()
                local waited = 0
                while CarryState.IsCarrying and waited < 1 do
                    waited += RunService.Heartbeat:Wait()
                end
            end

            enableFpsCap()
            setStatus(Text.GoEgg)

            if not runToEgg(uid, eggPos) then
                disableFpsCap()
                if CarryState.Cancelled then return end
                setStatus(Text.CantReach)
                return
            end
            if CarryState.Cancelled then disableFpsCap(); return end

            setStatus(Text.GrabEgg)
            if not grabEgg(uid, 4) then
                disableFpsCap()
                if CarryState.Cancelled then return end
                setStatus(Text.CantGrab)
                return
            end

            local delivered = deliverByTP(uid)
            if CarryState.Cancelled then return end
            setStatus(delivered and Text.DeliverOk or Text.DeliverFail)
        end)

        disableFpsCap()
        destroyClone()
        if not ok then setStatus(Text.Err .. tostring(err)) end
        CarryState.Busy = false
        if onFinished then onFinished() end
    end)
end

local function bayFlyHome()
    if AutoBay.Busy then return end
    if not CarryState.IsCarrying then return end
    local uid = CarryState.EggUid
    if type(uid) ~= "string" or uid == "" then return end

    AutoBay.Busy      = true
    AutoBay.Cancelled = false

    task.spawn(function()
        setStatus(Text.BayFlying)
        local ok, err = pcall(function()
            local delivered = deliverByTP(uid, function() return AutoBay.Cancelled end)
            setStatus(delivered and Text.BayDone or Text.BayFail)
        end)
        destroyClone()
        disableFpsCap()
        if not ok then setStatus(Text.Err .. tostring(err)) end
        AutoBay.Busy = false
    end)
end

local function startAutoBay()
    if AutoBay.Active then return end
    AutoBay.Active    = true
    AutoBay.Cancelled = false
    AutoBay.Thread    = task.spawn(function()
        local wasCarrying = false
        while AutoBay.Active do
            task.wait(0.3)
            local nowCarrying = CarryState.IsCarrying
            local shouldFire  = nowCarrying
                and not wasCarrying
                and not AutoBay.Busy
                and not CarryState.Busy
            wasCarrying = nowCarrying
            if shouldFire then
                task.wait(0.05)
                if AutoBay.Active
                and CarryState.IsCarrying
                and not AutoBay.Busy
                and not CarryState.Busy
                and type(CarryState.EggUid) == "string"
                and CarryState.EggUid ~= ""
                then
                    bayFlyHome()
                end
            end
        end
    end)
end

local function stopAutoBay()
    AutoBay.Active    = false
    AutoBay.Cancelled = true
    AutoBay.Busy      = false
    AutoBay.Thread    = nil
end

local function getSortedEggs()
    local list   = {}
    local ok, rc = pcall(getSnapshot)
    if not ok then return list end
    for _, rec in ipairs(rc) do list[#list + 1] = describeEgg(rec) end
    table.sort(list, function(a, b) return a.Value > b.Value end)
    return list
end

local function startAutoSteal()
    if AutoSteal.Active then return end
    AutoSteal.Active = true
    AutoSteal.Thread = task.spawn(function()
        while AutoSteal.Active do
            CarryState.Cancelled = false
            if CarryState.Busy then task.wait(0.3); continue end

            local eggs = getSortedEggs()
            if #eggs == 0 then
                setStatus(Text.AutoNoEgg)
                task.wait(3)
                continue
            end

            local target = eggs[1]
            setStatus(Text.AutoTaking .. target.Name)

            local done = false
            stealAndDeliver(target.Uid, function() done = true end)

            local bwait = 0
            while not CarryState.Busy and bwait < 0.5 and AutoSteal.Active do
                bwait += task.wait(0.05)
            end

            local timeout = 0
            while not done and AutoSteal.Active do
                task.wait(0.1)
                timeout += 0.1
                if timeout > 60 then
                    CarryState.Cancelled = true
                    local drain = 0
                    while CarryState.Busy and drain < 2 do drain += task.wait(0.1) end
                    CarryState.Busy = false
                    break
                end
            end

            task.wait(0.5)
        end
    end)
end

local function stopAutoSteal()
    AutoSteal.Active     = false
    AutoSteal.Thread     = nil
    CarryState.Cancelled = true
end

local statusWhitelist = {}

setStatus = function(msg)
    msg = tostring(msg)
    local prefix6 = string.sub(msg, 1, 6)
    local prefix5 = string.sub(msg, 1, 5)
    local prefix9 = string.sub(msg, 1, 9)
    local prefix3 = string.sub(msg, 1, 3)
    local prefix4 = string.sub(msg, 1, 4)
    if statusWhitelist[msg]
    or prefix6 == "Chọn:"
    or prefix5 == "Auto:"
    or prefix9 == "Auto Bay:"
    or prefix3 == "TP "
    or prefix3 == "Bị "
    or prefix4 == "Hạ c"
    or prefix3 == "Lỗi"
    then
        statusLabel.Text = msg
    elseif CarryState.Busy then
        statusLabel.Text = Text.DeliveringAlt
    elseif AutoBay.Busy then
        statusLabel.Text = Text.FlyingHomeAlt
    end
end

-- egg rows
local selectedUid = nil
local eggRows     = {}

local function setSelected(uid)
    selectedUid = uid
    for rowUid, row in pairs(eggRows) do
        local active    = rowUid == uid
        row.Stroke.Thickness = active and 1.5 or 1
        row.Gradient.Color   = active
            and makeGradient(White, DarkGrey)
            or  makeGradient(DarkGrey, Color3.fromRGB(30, 30, 34))
    end
end

local function makeEggRow(info, order)
    local frame = newInstance("TextButton", {
        Size             = UDim2.new(1, 0, 0, 28),
        BackgroundColor3 = Black,
        LayoutOrder      = order,
        Text             = "",
        AutoButtonColor  = true,

    frame.Parent = eggList

    newInstance("ImageLabel", {

    })

    addLabel(frame, {
        Size           = UDim2.new(1, -30, 0, 11),
        Position       = UDim2.fromOffset(28, 15),
        Font           = Enum.Font.GothamMedium,
        TextSize       = 9,
        TextXAlignment = Enum.TextXAlignment.Left,
        Text           = formatMoney(info.Value),
    })

    frame.MouseButton1Click:Connect(function()
        setSelected(info.Uid)
        setStatus(Text.Choose .. info.Name)
    end)

    return { Frame = frame, Stroke = stroke, Gradient = gradient }
end

local function refreshEggList()
    local ok, records = pcall(getSnapshot)
    if not ok then return end

    local infoList = {}
    for _, rec in ipairs(records) do infoList[#infoList + 1] = describeEgg(rec) end
    table.sort(infoList, function(a, b) return a.Value > b.Value end)

    local liveUids = {}
    for _, info in ipairs(infoList) do liveUids[info.Uid] = info end

    for uid, row in pairs(eggRows) do
        if not liveUids[uid] then
            row.Frame:Destroy()
            eggRows[uid] = nil
        end
    end

    for idx, info in ipairs(infoList) do
        if not eggRows[info.Uid] then
            eggRows[info.Uid] = makeEggRow(info, idx)
        else
            eggRows[info.Uid].Frame.LayoutOrder = idx
        end
    end

    for i = #gradients, 1, -1 do
        if not gradients[i].Parent then table.remove(gradients, i) end
    end

    if selectedUid and not liveUids[selectedUid] then selectedUid = nil end
    setSelected(selectedUid)
    if #infoList == 0 then setStatus(Text.ChoosePet) end
end

-- drag
do
    local dragging, startWinPos, startInputPos
    header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch
        then
            dragging      = true
            startWinPos   = window.Position
            startInputPos = input.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (
            input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch
        ) then
            local delta = input.Position - startInputPos
            window.Position = UDim2.new(
                startWinPos.X.Scale, startWinPos.X.Offset + delta.X,
                startWinPos.Y.Scale, startWinPos.Y.Offset + delta.Y
            )
        end
    end)
end

-- ============================================================
-- PROXIMITY PROMPT  (HoldDuration = 0, auto-cleanup on close)
-- ============================================================

local function applyProximityFix()
    for _, v in ipairs(workspace:GetDescendants()) do
        if v.ClassName == "ProximityPrompt" then
            pcall(function() v.HoldDuration = 0 end)
        end
    end
end

local proximityConn = nil

local function startProximityFix()
    applyProximityFix()
    proximityConn = workspace.DescendantAdded:Connect(function(v)
        if v.ClassName == "ProximityPrompt" then
            pcall(function() v.HoldDuration = 0 end)
        end
    end)
end

local function stopProximityFix()
    if proximityConn then proximityConn:Disconnect(); proximityConn = nil end
end

startProximityFix()
